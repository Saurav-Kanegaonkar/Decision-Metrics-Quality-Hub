SQL pattern

Metric trust scoring logic

select
  metric_name,
  metric_owner,
  definition_score,
  warehouse_variance_pct,
  excel_mismatch_rate,
  case
    when definition_score < 80 then 'definition drift'
    when warehouse_variance_pct > 4 then 'source variance'
    when excel_mismatch_rate > 4 then 'spreadsheet mismatch'
    else 'monitor'
  end as primary_quality_flag
from metric_quality_audit
where decision_criticality >= 4;
